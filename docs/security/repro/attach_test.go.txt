package attach

import (
	"os"
	"path/filepath"
	"testing"

	"github.com/piconic-ai/edit/internal/protocol"
)

// Run this test with `umask 077` in the invoking shell.
func TestSecurityAuditAttachmentPermissions(t *testing.T) {
	dir := t.TempDir()
	control := filepath.Join(dir, "control.txt")
	if err := os.WriteFile(control, []byte("control"), 0o666); err != nil {
		t.Fatal(err)
	}
	st, err := os.Stat(control)
	if err != nil {
		t.Fatal(err)
	}
	if st.Mode().Perm() != 0o600 {
		t.Skip("requires umask 077")
	}
	// An existing assets directory can already be traversable by other users.
	if err := os.Mkdir(filepath.Join(dir, Dir), 0o755); err != nil {
		t.Fatal(err)
	}
	if err := os.Chmod(filepath.Join(dir, Dir), 0o755); err != nil {
		t.Fatal(err)
	}
	a := &Attachments{opts: Options{File: filepath.Join(dir, "notes.md")}}
	data := []byte("synthetic attachment")
	rel, _, err := a.save(protocol.ContentHash(data), ".png", data)
	if err != nil {
		t.Fatal(err)
	}
	st, err = os.Stat(filepath.Join(dir, filepath.FromSlash(rel)))
	if err != nil {
		t.Fatal(err)
	}
	if st.Mode().Perm() != 0o644 {
		t.Fatalf("unexpected permissions: %o", st.Mode().Perm())
	}
	t.Logf("CONFIRMED: attachment mode=%o despite umask 077", st.Mode().Perm())
}
